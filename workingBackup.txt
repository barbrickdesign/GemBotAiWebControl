        // ==========================================
        // GemBot Controller - Dual Mode
        // ==========================================

        class GemBotController {
            constructor() {
                this.mode = 'browser';
                this.port = null;
                this.ws = null;
                this.isConnected = false;
                this.dataReceived = 0;
                this.dataSent = 0;
                this.availablePorts = [];
                this.connectionStartTime = null;
                this.messageLog = [];
                this.connectionAttempts = 0;
                this.lastErrorTime = null;
                
                // Enhanced diagnostics
                this.serialBuffer = '';  // Buffer for incomplete lines
                this.lastDataTime = null;  // Track last data reception
                this.bytesThisSecond = 0;  // For throughput calculation
                this.lastThroughputCheck = null;  // Throughput timer
                this.totalLinesReceived = 0;  // Track complete lines
                this.errorCount = 0;  // Track errors

                this.log('🚀 GemBot Controller initializing...', 'info');
                this.logBrowserInfo();
                this.initializeEventListeners();
                this.loadSettings();
                this.updateModeInfo();
                this.log('✅ GemBot Controller ready', 'success');
            }

            // ========== Enhanced Diagnostic Logging ==========
            log(message, type = 'log') {
                const timestamp = new Date().toLocaleTimeString('en-US', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' });
                const logEntry = { timestamp, message, type };
                this.messageLog.push(logEntry);

                // Console output with styling
                const styles = {
                    'error': 'color: #f44336; font-weight: bold; font-size: 13px;',
                    'warning': 'color: #ff9800; font-weight: bold; font-size: 13px;',
                    'success': 'color: #4caf50; font-weight: bold; font-size: 13px;',
                    'info': 'color: #2196F3; font-weight: bold; font-size: 13px;',
                    'debug': 'color: #9c27b0; font-size: 12px;',
                    'data': 'color: #607d8b; font-size: 12px;',
                    'diagnostic': 'color: #00bcd4; font-weight: bold; font-size: 13px;',
                    'throughput': 'color: #e91e63; font-weight: bold; font-size: 12px;'
                };

                // Enhanced console logging with visual hierarchy
                const prefix = type === 'error' ? '❌' : 
                              type === 'warning' ? '⚠️' :
                              type === 'success' ? '✅' :
                              type === 'info' ? 'ℹ️' :
                              type === 'debug' ? '🔍' :
                              type === 'diagnostic' ? '📊' :
                              type === 'throughput' ? '📈' : '•';
                
                console.log(`%c${prefix} [${timestamp}] ${message}`, styles[type] || 'color: #333;');

                // Keep only last 100 messages
                if (this.messageLog.length > 100) {
                    this.messageLog.shift();
                }
            }

            logBrowserInfo() {
                this.log('Browser: ' + (navigator.userAgent.match(/Chrome|Firefox|Safari|Edge/i)?.[0] || 'Unknown'), 'diagnostic');
                this.log('Platform: ' + navigator.platform, 'diagnostic');
                this.log('Serial API Available: ' + ('serial' in navigator), 'diagnostic');
                this.log('WebSocket Available: ' + (typeof WebSocket !== 'undefined'), 'diagnostic');
                this.log(`Memory: ${(navigator.deviceMemory || 'unknown')} GB`, 'diagnostic');
                this.log(`Cores: ${(navigator.hardwareConcurrency || 'unknown')}`, 'diagnostic');
                console.log('%c═════════════════════════════════════════════════════════════════', 'color: #667eea; font-weight: bold; font-size: 14px;');
                console.log('%c🚀 GemBot Serial Monitor Started', 'color: #667eea; font-weight: bold; font-size: 14px;');
                console.log('%c═════════════════════════════════════════════════════════════════', 'color: #667eea; font-weight: bold; font-size: 14px;');
                console.log('navigator.serial:', 'serial' in navigator);
                console.log('Browser Support: Web Serial API requires Chrome 89+, Edge 89+, or Opera 75+');
                console.log('Alternative: Use WebSocket Server mode for Firefox, Safari, or older browsers');
            }

            logConnectionDiagnostics(mode) {
                console.log('%c=== Connection Diagnostics ===', 'color: #667eea; font-weight: bold;');
                this.log(`Mode: ${mode === 'browser' ? 'Browser Serial API' : 'WebSocket Server'}`, 'diagnostic');
                this.log(`Attempt #${++this.connectionAttempts}`, 'diagnostic');
                this.log(`Time: ${new Date().toLocaleString()}`, 'diagnostic');

                if (mode === 'browser') {
                    const portSelect = document.getElementById('portSelect');
                    const selectedIndex = portSelect.value;
                    this.log(`Ports detected: ${this.availablePorts.length}`, 'diagnostic');
                    this.log(`Selected port index: ${selectedIndex || 'none'}`, 'diagnostic');
                    if (selectedIndex !== '' && this.availablePorts[selectedIndex]) {
                        const info = this.availablePorts[selectedIndex].getInfo?.();
                        if (info) {
                            this.log(`USB VID: 0x${(info.usbVendorId || 0).toString(16).toUpperCase().padStart(4, '0')} PID: 0x${(info.usbProductId || 0).toString(16).toUpperCase().padStart(4, '0')}`, 'diagnostic');
                        }
                    }
                } else {
                    const serverUrl = document.getElementById('serverInput').value;
                    this.log(`Server: ${serverUrl}`, 'diagnostic');
                    this.log(`Protocol: ${serverUrl.includes('https') || serverUrl.includes('wss') ? 'WSS' : 'WS'}`, 'diagnostic');
                }
            }

            initializeEventListeners() {
                // Mode switching
                document.querySelectorAll('input[name="mode"]').forEach(radio => {
                    radio.addEventListener('change', (e) => this.switchMode(e.target.value));
                });

                // Connection Controls
                document.getElementById('scanButton').addEventListener('click', () => this.scanPorts());
                document.getElementById('connectButton').addEventListener('click', () => this.connect());
                document.getElementById('disconnectButton').addEventListener('click', () => this.disconnect());

                // Control Buttons
                document.querySelectorAll('.btn-direction, .btn-control').forEach(btn => {
                    if (!btn.id.includes('Clear') && !btn.id.includes('Scan') && 
                        !btn.id.includes('Connect') && !btn.id.includes('Disconnect')) {
                        btn.addEventListener('click', () => this.handleControlButton(btn));
                    }
                });

                // Serial Input
                document.getElementById('serialInput').addEventListener('keypress', (e) => {
                    if (e.key === 'Enter') this.sendSerialCommand();
                });
                document.getElementById('sendButton').addEventListener('click', () => this.sendSerialCommand());
                document.getElementById('btnClearMonitor').addEventListener('click', () => this.clearMonitor());

                // Quick commands
                this.setupQuickCommands();
            }

            switchMode(newMode) {
                this.mode = newMode;
                this.saveSettings();

                // Update UI
                document.querySelectorAll('.mode-specific').forEach(el => el.classList.remove('active'));
                document.querySelectorAll(`.${newMode}-mode`).forEach(el => el.classList.add('active'));

                this.updateModeInfo();
                this.addMessage('info', `Switched to ${newMode === 'browser' ? 'Browser Serial API' : 'WebSocket Server'} mode`);

                if (this.isConnected) {
                    this.disconnect();
                }
            }

            updateModeInfo() {
                const infoEl = document.getElementById('modeInfo');
                const warningEl = document.getElementById('modeWarning');
                const modeStatusEl = document.getElementById('statusMode');

                if (this.mode === 'browser') {
                    infoEl.innerHTML = '✅ Direct USB connection - No server needed';
                    warningEl.innerHTML = 'Requires Chrome, Edge, or Opera. USB device must be connected to this computer.';
                    modeStatusEl.textContent = 'Browser API';
                    document.getElementById('statusMode').style.color = '#4caf50';
                } else {
                    infoEl.innerHTML = '✅ Network bridge - Works on any device';
                    warningEl.innerHTML = 'Connect to Node.js server running on your network. Supports remote access.';
                    modeStatusEl.textContent = 'WebSocket';
                    document.getElementById('statusMode').style.color = '#2196F3';
                }
            }

            setupQuickCommands() {
                const commands = {
                    'btnSettings': 'page 0\xFF\xFF\xFF',
                    'btnDesign': 'page 14\xFF\xFF\xFF',
                    'btnPreform': 'page 16\xFF\xFF\xFF',
                    'btnCut': 'page 17\xFF\xFF\xFF',
                    'btnPolish': 'page 18\xFF\xFF\xFF',
                    'btnCalibrate': 'Your calibration command here'
                };

                Object.entries(commands).forEach(([btnId, cmd]) => {
                    const btn = document.getElementById(btnId);
                    if (btn) {
                        btn.addEventListener('click', () => {
                            this.sendToDevice(cmd);
                        });
                    }
                });
            }

            handleControlButton(btn) {
                if (!this.isConnected) {
                    this.addMessage('error', 'Device not connected!');
                    return;
                }

                const cmd = btn.dataset.cmd;
                if (cmd) {
                    this.sendToDevice(cmd);
                    this.addMessage('command', `→ ${btn.textContent.trim()}`);
                }
            }

            // ========== Browser Mode ==========

            async scanPorts() {
                if (this.mode !== 'browser') return;

                if (!('serial' in navigator)) {
                    this.addMessage('warning', 'Serial API not available');
                    this.log('ERROR: navigator.serial is undefined. Browser does not support Web Serial API.', 'error');
                    this.log('Solution: Use Chrome 89+, Edge 89+, Opera 75+, or switch to WebSocket Server mode', 'info');
                    return;
                }

                try {
                    document.getElementById('scanButton').disabled = true;
                    this.log('🔍 Starting port scan...', 'info');
                    this.addMessage('info', 'Scanning ports...');

                    const startTime = performance.now();
                    const ports = await navigator.serial.getPorts();
                    const scanTime = (performance.now() - startTime).toFixed(2);

                    this.availablePorts = ports;
                    this.log(`Port scan completed in ${scanTime}ms`, 'diagnostic');

                    const portSelect = document.getElementById('portSelect');
                    portSelect.innerHTML = '<option value="">Select COM Port...</option>';

                    if (ports.length === 0) {
                        this.log('⚠️ No USB devices detected', 'warning');
                        this.addMessage('warning', 'No USB devices found. Troubleshooting steps:\n1. Check USB cable is connected\n2. Check Device Manager for COM ports\n3. Install Arduino drivers if missing\n4. Try different USB port\n5. Close Arduino IDE if open');
                    } else {
                        this.log(`✅ Found ${ports.length} device(s)`, 'success');
                        ports.forEach((port, index) => {
                            const option = document.createElement('option');
                            option.value = index;
                            try {
                                const info = port.getInfo?.();
                                const vendorId = info?.usbVendorId ? `0x${info.usbVendorId.toString(16).toUpperCase().padStart(4, '0')}` : 'Unknown';
                                option.textContent = `COM${index + 1} (VID: ${vendorId})`; 
                                this.log(`  Port ${index + 1}: VID=0x${info?.usbVendorId?.toString(16).toUpperCase().padStart(4, '0') || '????'}, PID=0x${info?.usbProductId?.toString(16).toUpperCase().padStart(4, '0') || '????'}`, 'diagnostic');
                            } catch (e) {
                                option.textContent = `COM${index + 1}`;
                                this.log(`  Port ${index + 1}: (info unavailable)`, 'diagnostic');
                            }
                            portSelect.appendChild(option);
                        });
                        this.addMessage('success', `Found ${ports.length} device(s) - Select port and click Connect`);
                    }
                } catch (error) {
                    this.lastErrorTime = Date.now();
                    this.log(`Scan error: ${error.message}`, 'error');
                    this.log(`Error code: ${error.code || 'N/A'}`, 'debug');
                    this.addMessage('error', `Scan error: ${error.message}\n\nTroubleshooting:\n• Close Arduino IDE\n• Unplug and replug USB cable\n• Try different USB port\n• Restart browser`);
                } finally {
                    document.getElementById('scanButton').disabled = false;
                }
            }

            async connect() {
                if (this.mode === 'browser') {
                    await this.connectBrowser();
                } else {
                    await this.connectServer();
                }
            }

            async connectBrowser() {
                if (!('serial' in navigator)) {
                    this.addMessage('error', 'Serial API not supported');
                    this.log('ERROR: Web Serial API not available in this browser', 'error');
                    return;
                }

                this.logConnectionDiagnostics('browser');
                this.connectionStartTime = Date.now();

                const portIndex = document.getElementById('portSelect').value;
                if (portIndex === '') {
                    try {
                        this.log('Requesting port selection from user...', 'info');
                        this.port = await navigator.serial.requestPort();
                        this.availablePorts.push(this.port);
                        this.log('User selected a port', 'success');
                    } catch (error) {
                        this.log(`Port selection cancelled by user: ${error.message}`, 'warning');
                        this.addMessage('warning', `Port request cancelled`);
                        return;
                    }
                } else {
                    this.port = this.availablePorts[portIndex];
                    this.log(`Selected port index: ${portIndex}`, 'info');
                }

                try {
                    this.log('Attempting to open port with baudRate 9600...', 'info');
                    const openStartTime = performance.now();
                    
                    await this.port.open({ baudRate: 9600 });
                    
                    const openTime = (performance.now() - openStartTime).toFixed(2);
                    this.log(`✅ Port opened successfully in ${openTime}ms`, 'success');
                    
                    this.isConnected = true;
                    this.dataReceived = 0;
                    this.dataSent = 0;
                    
                    const connTime = ((Date.now() - this.connectionStartTime) / 1000).toFixed(2);
                    this.log(`Connection established in ${connTime}s`, 'success');
                    
                    this.updateUI();
                    this.addMessage('success', `✓ Connected successfully (${connTime}s)`);
                    this.startReading();
                    
                    console.log('%c=== Connection Successful ===', 'color: #4caf50; font-weight: bold;');
                    console.log('Port is ready for communication');
                    console.log('Baud rate: 9600');
                    console.log('Ready to send/receive data');
                    
                } catch (error) {
                    this.lastErrorTime = Date.now();
                    const connTime = ((Date.now() - this.connectionStartTime) / 1000).toFixed(2);
                    
                    this.log(`❌ Connection FAILED after ${connTime}s`, 'error');
                    this.log(`Error: ${error.message}`, 'error');
                    this.log(`Error name: ${error.name}`, 'debug');
                    
                    let troubleshootingMsg = '';
                    if (error.message.includes('open') || error.message.includes('busy')) {
                        troubleshootingMsg = '\nTroubleshooting:\n• Port may be in use by another application\n• Close Arduino IDE Serial Monitor\n• Close PuTTY or other serial tools\n• Unplug and replug USB cable';
                    } else if (error.message.includes('permission')) {
                        troubleshootingMsg = '\nTroubleshooting:\n• Browser may not have permission\n• Try a different browser (Chrome/Edge)\n• Check Windows Device Manager';
                    }
                    
                    this.addMessage('error', `Connection failed: ${error.message}${troubleshootingMsg}`);
                }
            }

            // ========== Server Mode ==========

            async connectServer() {
                const serverUrl = document.getElementById('serverInput').value;
                if (!serverUrl) {
                    this.log('No server URL provided', 'warning');
                    this.addMessage('warning', 'Enter server address (e.g., localhost:3000)');
                    return;
                }

                this.logConnectionDiagnostics('server');
                this.connectionStartTime = Date.now();

                try {
                    const wsProtocol = serverUrl.includes('https') || serverUrl.includes('wss') ? 'wss' : 'ws';
                    const url = serverUrl.includes('://') ? serverUrl : `${wsProtocol}://${serverUrl}`;

                    this.log(`Resolving server: ${url}`, 'info');
                    this.addMessage('info', `Connecting to ${serverUrl}...`);

                    // Create WebSocket with diagnostics
                    this.ws = new WebSocket(url);
                    this.log(`WebSocket state: ${this.ws.readyState} (0=connecting)`, 'diagnostic');

                    this.ws.onopen = () => {
                        const connTime = ((Date.now() - this.connectionStartTime) / 1000).toFixed(2);
                        this.isConnected = true;
                        this.dataReceived = 0;
                        this.dataSent = 0;
                        
                        this.log(`✅ WebSocket connected in ${connTime}s`, 'success');
                        this.log(`WebSocket state: ${this.ws.readyState} (1=open)`, 'diagnostic');
                        this.log(`Server URL: ${url}`, 'diagnostic');
                        
                        this.updateUI();
                        this.addMessage('success', `✓ Connected to server (${connTime}s)`);
                        document.getElementById('statusPort').textContent = serverUrl;
                        
                        console.log('%c=== WebSocket Connection Successful ===', 'color: #4caf50; font-weight: bold;');
                        console.log('Connected to:', url);
                        console.log('Ready to send/receive data');
                    };

                    this.ws.onmessage = (event) => {
                        try {
                            const message = JSON.parse(event.data);
                            this.log(`Received: ${message.type}`, 'data');
                            this.handleServerMessage(message);
                        } catch (parseError) {
                            this.log(`JSON parse error: ${parseError.message}`, 'error');
                            this.log(`Raw data: ${event.data.substring(0, 100)}...`, 'debug');
                        }
                    };

                    this.ws.onerror = (error) => {
                        this.lastErrorTime = Date.now();
                        this.log(`WebSocket error occurred`, 'error');
                        this.log(`Error type: ${error.type || 'unknown'}`, 'debug');
                        this.log(`WebSocket state: ${this.ws?.readyState}`, 'diagnostic');
                        
                        let errorMsg = 'Connection error';
                        if (this.ws?.readyState === WebSocket.CLOSED) {
                            errorMsg = 'Connection refused - server may not be running';
                            this.log('Server connection refused', 'error');
                            this.log('Solutions: 1) Run npm start, 2) Check port number, 3) Check firewall', 'info');
                        }
                        
                        this.addMessage('error', errorMsg);
                    };

                    this.ws.onclose = () => {
                        this.isConnected = false;
                        this.log(`WebSocket connection closed (state: ${this.ws?.readyState})`, 'info');
                        this.updateUI();
                        this.addMessage('info', 'Disconnected from server');
                    };

                    // Connection timeout
                    setTimeout(() => {
                        if (this.ws && this.ws.readyState === WebSocket.CONNECTING) {
                            const connTime = ((Date.now() - this.connectionStartTime) / 1000).toFixed(2);
                            this.log(`❌ Connection TIMEOUT after ${connTime}s`, 'error');
                            this.log('WebSocket stuck in CONNECTING state', 'debug');
                            this.log('Possible causes: 1) Server not running, 2) Wrong URL, 3) Firewall blocking, 4) Network issue', 'info');
                            
                            this.addMessage('error', `Connection timeout (${connTime}s)\n\nTroubleshooting:\n1. Is server running? (npm start)\n2. Is URL correct? (localhost:3000)\n3. Check Windows Firewall\n4. Check npm console for errors`);
                            this.ws.close();
                            this.ws = null;
                        }
                    }, 5000);

                } catch (error) {
                    this.lastErrorTime = Date.now();
                    this.log(`Error creating WebSocket: ${error.message}`, 'error');
                    this.addMessage('error', `Error: ${error.message}`);
                }
            }

            handleServerMessage(message) {
                const { type, data } = message;

                switch (type) {
                    case 'data':
                        this.dataReceived += data.content.length;
                        document.getElementById('statusDataCount').textContent = `${this.dataReceived} bytes`;
                        this.addMessage('data', data.content);
                        break;
                    case 'connection':
                        if (data.status === 'connected') {
                            this.addMessage('success', 'Device connected');
                        } else {
                            this.addMessage('warning', 'Device disconnected');
                        }
                        break;
                    case 'error':
                        this.addMessage('error', data.message);
                        break;
                }
            }

            async startReading() {
                if (this.mode !== 'browser' || !this.port) return;

                try {
                    while (this.port && this.port.readable) {
                        const reader = this.port.readable.getReader();
                        try {
                            while (true) {
                                const { value, done } = await reader.read();
                                if (done) break;

                                const text = new TextDecoder().decode(value);
                                this.dataReceived += value.length;
                                this.processSerialData(text);  // Use enhanced processing
                                this.updateThroughput();
                                document.getElementById('statusDataCount').textContent = `${this.dataReceived} bytes (${this.totalLinesReceived} lines)`;
                            }
                        } finally {
                            reader.releaseLock();
                        }
                    }
                } catch (error) {
                    if (error.name !== 'NetworkError') {
                        this.errorCount++;
                        this.addMessage('error', `Read error: ${error.message}`);
                        this.log(`Read error (${this.errorCount}): ${error.message}`, 'error');
                    }
                }
            }
            
            processSerialData(rawData) {
                // Add raw data to buffer
                this.serialBuffer += rawData;
                this.lastDataTime = Date.now();
                this.bytesThisSecond += rawData.length;
                
                // Split on newlines and process complete lines
                const lines = this.serialBuffer.split('\n');
                
                // Keep the last incomplete line in buffer
                this.serialBuffer = lines[lines.length - 1];
                
                // Process all complete lines
                for (let i = 0; i < lines.length - 1; i++) {
                    const line = lines[i].trim();
                    if (line.length > 0) {
                        this.totalLinesReceived++;
                        this.addMessage('data', line);  // Display complete line
                        this.log(`[RX-Line ${this.totalLinesReceived}] ${this.sanitizeForLog(line)}`, 'data');
                    }
                }
            }
            
            updateThroughput() {
                // Reset throughput counter every second
                if (!this.lastThroughputCheck) {
                    this.lastThroughputCheck = Date.now();
                } else if (Date.now() - this.lastThroughputCheck >= 1000) {
                    this.log(`Throughput: ${this.bytesThisSecond} bytes/sec (${this.totalLinesReceived} lines total)`, 'throughput');
                    this.bytesThisSecond = 0;
                    this.lastThroughputCheck = Date.now();
                }
            }
            
            sanitizeForLog(data) {
                // Make special characters visible in logs
                return data.replace(/[\x00-\x1F]/g, (c) => {
                    const codes = { '\x00': '\\0', '\n': '\\n', '\r': '\\r', '\t': '\\t', '\x1b': 'ESC' };
                    return codes[c] || `[${c.charCodeAt(0).toString(16).padStart(2, '0').toUpperCase()}]`;
                });
            }

            async sendToDevice(data) {
                if (!this.isConnected) {
                    this.log('Cannot send: device not connected', 'warning');
                    this.addMessage('error', 'Not connected');
                    return false;
                }

                try {
                    this.dataSent += data.length;

                    if (this.mode === 'browser' && this.port && this.port.writable) {
                        const writer = this.port.writable.getWriter();
                        await writer.write(new TextEncoder().encode(data));
                        writer.releaseLock();
                        this.log(`[SEND] ${data.length} bytes: ${this.sanitizeData(data)}`, 'data');
                    } else if (this.mode === 'server' && this.ws && this.ws.readyState === WebSocket.OPEN) {
                        const payload = JSON.stringify({
                            command: 'send',
                            data: { content: data },
                            timestamp: new Date().toISOString()
                        });
                        this.ws.send(payload);
                        this.log(`[SEND-WS] ${data.length} bytes: ${this.sanitizeData(data)}`, 'data');
                    } else {
                        this.log('Send failed: no active connection', 'error');
                        return false;
                    }
                    
                    document.getElementById('statusDataCount').textContent = `RX: ${this.dataReceived} | TX: ${this.dataSent}`;
                    return true;
                } catch (error) {
                    this.log(`Send error: ${error.message}`, 'error');
                    this.addMessage('error', `Send failed: ${error.message}`);
                    return false;
                }
            }

            sanitizeData(data) {
                // Make binary/special chars visible in logs
                return data.replace(/[\x00-\x1F]/g, (c) => {
                    const codes = { '\x00': '\\0', '\n': '\\n', '\r': '\\r', '\t': '\\t', '\x1b': 'ESC' };
                    return codes[c] || `[${c.charCodeAt(0).toString(16).padStart(2, '0').toUpperCase()}]`;
                });
            }

            async sendSerialCommand() {
                const input = document.getElementById('serialInput');
                const command = input.value.trim();
                if (!command) return;

                if (await this.sendToDevice(command)) {
                    this.addMessage('command', `→ ${command}`);
                    input.value = '';
                }
            }

            async disconnect() {
                try {
                    const connTime = this.connectionStartTime ? ((Date.now() - this.connectionStartTime) / 1000).toFixed(2) : 'unknown';
                    
                    if (this.mode === 'browser' && this.port) {
                        await this.port.close();
                        this.port = null;
                        this.log(`Closed serial port (connected for ${connTime}s)`, 'info');
                    } else if (this.mode === 'server' && this.ws) {
                        this.ws.close();
                        this.ws = null;
                        this.log(`Closed WebSocket connection (connected for ${connTime}s)`, 'info');
                    }
                    
                    this.log(`Total data RX: ${this.dataReceived} bytes, TX: ${this.dataSent} bytes`, 'diagnostic');
                    this.isConnected = false;
                    this.connectionStartTime = null;
                    this.updateUI();
                    this.addMessage('info', 'Disconnected');
                } catch (error) {
                    this.log(`Disconnect error: ${error.message}`, 'error');
                    this.addMessage('error', `Disconnect error: ${error.message}`);
                }
            }

            updateUI() {
                const isConnected = this.isConnected;
                document.getElementById('connectButton').classList.toggle('hidden', isConnected);
                document.getElementById('disconnectButton').classList.toggle('hidden', !isConnected);
                document.getElementById('statusIndicator').classList.toggle('connected', isConnected);
                document.getElementById('statusConnection').textContent = isConnected ? 'Connected' : 'Disconnected';
                document.getElementById('serialInput').disabled = !isConnected;
                document.getElementById('sendButton').disabled = !isConnected;
                document.getElementById('scanButton').disabled = isConnected;
                document.getElementById('portSelect').disabled = isConnected;
                document.getElementById('serverInput').disabled = isConnected;
            }

            clearMonitor() {
                document.getElementById('serialMonitor').textContent = '';
                this.dataReceived = 0;
                document.getElementById('statusDataCount').textContent = '0 bytes';
            }

            addMessage(type, message) {
                const monitor = document.getElementById('serialMonitor');
                const timestamp = new Date().toLocaleTimeString('en-US', { hour12: false });

                const icons = {
                    'data': '⬅',
                    'command': '→',
                    'info': '●',
                    'success': '✓',
                    'warning': '⚠',
                    'error': '✗',
                    'debug': '🔍'
                };
                
                const prefix = icons[type] || '•';
                const line = `[${timestamp}] [${prefix}] ${message}\n`;
                monitor.textContent += line;
                monitor.scrollTop = monitor.scrollHeight;
                
                // Also log to console
                this.log(`UI: ${message}`, type);
            }

                localStorage.setItem('gembot-server', document.getElementById('serverInput').value);
            }

                document.querySelector(`input[value="${mode}"]`).checked = true;
                document.getElementById('serverInput').value = server;
                this.mode = mode;
            }
        }

            window.gemBotController = new GemBotController();

        window.addEventListener('beforeunload', (e) => {

                e.preventDefault();

            }
        });
